ADD POSTGRESQL DATABASE AND BACKEND API TO the PROJECT

DATABASE SETUP:
• Create PostgreSQL database with these tables:

-- Main application table
applications (
  id SERIAL PRIMARY KEY,
  first_name TEXT NOT NULL,
  last_name TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE,
  phone TEXT,
  years_experience TEXT,
  how_heard_about_us TEXT,
  submitted_at TIMESTAMPTZ DEFAULT NOW()
)

-- Areas of interest (many-to-many relationship)
interest_areas (
  id SERIAL PRIMARY KEY,
  name TEXT NOT NULL UNIQUE
)

application_interests (
  id SERIAL PRIMARY KEY,
  application_id INT REFERENCES applications(id) ON DELETE CASCADE,
  interest_area_id INT REFERENCES interest_areas(id)
)

-- Skills (many-to-many relationship)  
skills (
  id SERIAL PRIMARY KEY,
  name TEXT NOT NULL UNIQUE,
  category TEXT NOT NULL -- 'Technical Skills', 'Soft Skills', 'Industry Knowledge'
)

application_skills (
  id SERIAL PRIMARY KEY,
  application_id INT REFERENCES applications(id) ON DELETE CASCADE,
  skill_id INT REFERENCES skills(id)
)
